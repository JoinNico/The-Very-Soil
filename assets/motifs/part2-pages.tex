% Vector layout for the Part II divider page.
% Loaded by config/theme.tex after book-motifs and part1-pages.  The shared
% \BookFourPointStar is defined in part1-pages.tex, which loads first.
%
% The page furniture follows Part I, while the centre opens its closed time ring
% into an asymmetric network of parallel worlds.  Every mark remains vector so
% the divider can be printed without a placed raster illustration.

\newcommand{\BookPartTwoPage}[1]{%
  \begin{tikzpicture}[
    remember picture,overlay,line cap=round,line join=round,
    part route/.style={HomuraViolet,opacity=.68,line width=.42pt},
    part route pink/.style={MadokaPink,opacity=.64,line width=.38pt},
    part orbit/.style={HomuraViolet,opacity=.42,line width=.3pt,densely dashed},
    part trace/.style={MadokaPink,opacity=.48,line width=.28pt,densely dotted},
    part world/.style={circle,draw=HomuraViolet,fill=white,
      line width=.55pt,inner sep=0pt},
    part junction/.style={circle,draw=HomuraViolet,fill=white,
      line width=.38pt,inner sep=0pt,minimum size=1.5mm}
  ]
    % Part pages are recto.  Their visual axis follows the bound text block,
    % whose centre is shifted outwards from the physical paper centre.
    \coordinate (PartCenter) at
      ($(current page.center)+(\BookRectoTextCenterShift,0)$);
    \coordinate (PartNorth) at (PartCenter |- current page.north);
    \coordinate (PartSouth) at (PartCenter |- current page.south);
    \coordinate (CenterMotif) at ($(PartCenter)+(0,-10mm)$);

    % Keep the fore-edge inset at 4.5 mm, then mirror it around the recto
    % visual axis.  With the current layout this leaves 11.5 mm at the spine.
    \coordinate (PartFrameNE) at
      ($(current page.north east)+(-4.5mm,-4.5mm)$);
    \coordinate (PartFrameSE) at
      ($(current page.south east)+(-4.5mm,4.5mm)$);
    \coordinate (PartFrameTopCenter) at ($(PartNorth)+(0,-4.5mm)$);
    \coordinate (PartFrameBottomCenter) at ($(PartSouth)+(0,4.5mm)$);
    \coordinate (PartFrameNW) at ($(PartFrameTopCenter)!-1!(PartFrameNE)$);
    \coordinate (PartFrameSW) at ($(PartFrameBottomCenter)!-1!(PartFrameSE)$);

    % Hairline magical boundary, corner stars, and short dotted traces.
    \draw[MadokaPink,opacity=.42,line width=.28pt,rounded corners=2mm]
      (PartFrameNW) rectangle (PartFrameSE);
    \coordinate (PartNEStar) at ($(current page.north east)+(-7mm,-7mm)$);
    \coordinate (PartSEStar) at ($(current page.south east)+(-7mm,7mm)$);
    \coordinate (PartTopStarCenter) at ($(PartNorth)+(0,-7mm)$);
    \coordinate (PartBottomStarCenter) at ($(PartSouth)+(0,7mm)$);
    \coordinate (PartNWStar) at ($(PartTopStarCenter)!-1!(PartNEStar)$);
    \coordinate (PartSWStar) at ($(PartBottomStarCenter)!-1!(PartSEStar)$);
    \foreach \corner in {PartNWStar,PartNEStar,PartSWStar,PartSEStar}{%
      \begin{scope}[shift={(\corner)},scale=.135]
        \path[fill=MadokaPink,fill opacity=.82] \BookFourPointStar
      \end{scope}
    }
    \draw[MadokaPink,opacity=.56,line width=.3pt,densely dotted]
      ($(PartNWStar)+(3mm,0)$)--($(PartNWStar)+(12mm,0)$)
      ($(PartNWStar)+(0,-3mm)$)--($(PartNWStar)+(0,-12mm)$)
      ($(PartNEStar)+(-3mm,0)$)--($(PartNEStar)+(-12mm,0)$)
      ($(PartNEStar)+(0,-3mm)$)--($(PartNEStar)+(0,-12mm)$)
      ($(PartSWStar)+(3mm,0)$)--($(PartSWStar)+(12mm,0)$)
      ($(PartSWStar)+(0,3mm)$)--($(PartSWStar)+(0,12mm)$)
      ($(PartSEStar)+(-3mm,0)$)--($(PartSEStar)+(-12mm,0)$)
      ($(PartSEStar)+(0,3mm)$)--($(PartSEStar)+(0,12mm)$);

      ($(TopEmblem)+(-36mm,0)$)--($(TopEmblem)+(-7mm,0)$)
      ($(TopEmblem)+(7mm,0)$)--($(TopEmblem)+(36mm,0)$);
    \foreach \x in {-37,37}{%
      \begin{scope}[shift={($(TopEmblem)+(\x mm,0)$)},scale=.06]
        \path[fill=MadokaPink] \BookFourPointStar

      at ($(PartNorth)+(0,-28mm)$) {PART II};

    \end{scope}

      ($(PartNorth)+(-25mm,-58mm)$)--($(PartNorth)+(25mm,-58mm)$);
    \node[anchor=center,text=MadokaPink,
      font=\rmfamily\fontsize{6.3pt}{8pt}\selectfont\addfontfeatures{LetterSpace=22}]
      at ($(PartNorth)+(0,-62.5mm)$) {SIDE STORY MANGA};

    % Narrow side notes: a Chinese branching maxim at the spine side and the
    % Part II conceptual triad at the fore edge.
    \coordinate (PartRightBand) at ($(current page.east)+(-8.2mm,0)$);
    \coordinate (PartLeftBand) at ($(PartCenter)!-1!(PartRightBand)$);
    \begin{scope}[shift={($(PartLeftBand)+(0,-11mm)$)},rotate=270,transform shape]

    \begin{scope}[shift={($(PartRightBand)+(0,11mm)$)},rotate=270,transform shape]

    \foreach \side in {PartLeftBand,PartRightBand}{%

    % one outgoing curve; each tail ends at its own coloured four-point mark.
    \coordinate (MidNW)  at ($(CenterMotif)+(-23.8mm,17.6mm)$);
    \coordinate (TailNW) at ($(CenterMotif)+(-37mm,22mm)$);
    \coordinate (EndNW)  at ($(CenterMotif)+(-43mm,25.5mm)$);

    \coordinate (TailW)  at ($(CenterMotif)+(-37.8mm,4.4mm)$);
    \coordinate (EndW)   at ($(CenterMotif)+(-43mm,1mm)$);
    \coordinate (MidSW)  at ($(CenterMotif)+(-25.5mm,-15mm)$);
    \coordinate (TailSW) at ($(CenterMotif)+(-37mm,-22mm)$);
    \coordinate (EndSW)  at ($(CenterMotif)+(-43mm,-27mm)$);
    \coordinate (MidS)   at ($(CenterMotif)+(-10.5mm,-22mm)$);
    \coordinate (TailS)  at ($(CenterMotif)+(-17.5mm,-30mm)$);
    \coordinate (EndS)   at ($(CenterMotif)+(-22mm,-34mm)$);

    \coordinate (EndE)   at ($(CenterMotif)+(43mm,9mm)$);
