% 新建一个计数器来控制 interlude 的编号
\newcounter{interludecounter}
\setcounter{interludecounter}{0} % 从1开始计数

% 定义新的命令 \interlude，显示为 "Interlude 数字 标题"，并且可以插入自定义标题
\newcommand{\interlude}[1]{ 
  \stepcounter{interludecounter} % 自增 interlude 编号
  \chapter*{Interlude \theinterludecounter\\[25pt] #1} % 标题显示为 "Interlude 数字: 自定义标题"
  \addcontentsline{toc}{chapter}{Interlude \theinterludecounter\ #1} % 将 Interlude 添加到目录中
  \markboth{Interlude}{Interlude} % 设置页眉
}

% 定义新的命令 \introduction，显示为 "Introduction" 然后换行显示标题，不带编号
\newcommand{\introduction}[1]{ 
  \chapter*{Introduction\\[25pt]#1} % 不带编号的 Introduction 章节，使用换行
  \addcontentsline{toc}{chapter}{Introduction\ #1} % 将 Introduction 添加到目录中
  \markboth{Introduction}{Introduction} % 设置页眉
}
