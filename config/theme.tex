% license have deliberately been added to assets/fonts/MadokaRunes/.
\newif\ifbookrunes
\IfFileExists{assets/fonts/MadokaRunes/MADOKARUNES-2.0.TTF}{%
  \IfFileExists{assets/fonts/MadokaRunes/LICENSE.txt}{%
    \bookrunestrue
    \newfontfamily\bookrunefont{MADOKARUNES-2.0.TTF}[
      Path=assets/fonts/MadokaRunes/,
      BoldFont=MADOKARUNES-2.0.TTF,
      ItalicFont=MADOKARUNES-2.0.TTF,
      BoldItalicFont=MADOKARUNES-2.0.TTF
    ]
  }{\bookrunesfalse}%
}{\bookrunesfalse}

\newcommand{\runeecho}[1]{%
  \ifbookrunes
    {\bookrunefont\mdseries\upshape #1}%
  \else
    {\sffamily\addfontfeatures{LetterSpace=18}\MakeUppercase{#1}}%
  \fi
}
