\newcommand{\interlude}[1]{%
  \stepcounter{interludecounter}%
  \booksimplechapterfalse
  \renewcommand{\BookSpecialKicker}{INTERLUDE}%
  \edef\BookSpecialNumber{\ifnum\value{interludecounter}<10 0\fi\arabic{interludecounter}}%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{%
    \protect\numberline{\protect\BookTOCLabel{间奏\Chinese{interludecounter}}}#1}%
  \markboth{间奏 \Chinese{interludecounter}\quad #1}{间奏 \Chinese{interludecounter}\quad #1}%
  \renewcommand{\BookSpecialKicker}{}%
  \renewcommand{\BookSpecialNumber}{}%
}

\newcommand{\introduction}[1]{%
  \booksimplechaptertrue
  \renewcommand{\BookSpecialKicker}{INTRODUCTION}%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{%
    \protect\numberline{\protect\BookTOCLabel{导言}}#1}%
  \markboth{导言\quad #1}{导言\quad #1}%
  \renewcommand{\BookSpecialKicker}{}%
  \booksimplechapterfalse
}

\newcommand{\bookbackchapter}[3][]{%
  \booksimplechaptertrue
  \renewcommand{\BookSpecialKicker}{#2}%
  \chapter*{#3}%
  \ifstrempty{#1}{%
    \addcontentsline{toc}{chapter}{#3}%
  }{%
    \addcontentsline{toc}{chapter}{%
      \protect\texorpdfstring{%
        \protect\numberline{\protect\BookTOCLabel{#1}}%
      }{#1}}%
  }%
  \markboth{#3}{#3}%
  \renewcommand{\BookSpecialKicker}{}%
  \booksimplechapterfalse
}

\newcommand{\bookpart}[1]{%
  \cleardoublepage
  \refstepcounter{part}%
  \phantomsection
  \ifnum\value{part}=2\relax
    \addtocontents{toc}{\protect\BookTOCUseHomura}%
  \fi
  \addcontentsline{toc}{part}{第\Chinese{part}部分\BookTitleSep #1}%
  \markboth{第\Chinese{part}部分\quad #1}{}%
  \thispagestyle{empty}%
  \ifcase\value{part}%
    % part counter is 0 only before the first \bookpart call; unreachable here.
  \or
    \BookPartOnePage{#1}%
  \or
    \BookPartTwoPage{#1}%
  \or
    \BookPartThreePage{#1}%
  \else
    \PackageError{the-very-soil}{No part page layout defined for part \arabic{part}}%
      {Add a BookPartFourPage command and a new case here.}%
  \fi
  \clearpage
  \thispagestyle{empty}\null\clearpage
}

% Future plate contract: exactly four single-page 131 x 182 mm bleed PDFs.
\newcommand{\BookRequireSinglePagePlate}[1]{%
  \IfFileExists{#1}{%
    \ifnum\XeTeXpdfpagecount "#1"=1\relax\else
      \PackageError{the-very-soil}{Plate #1 is not a single-page PDF}%
        {Split the artwork into page-01.pdf through page-04.pdf.}%
    \fi
  }{%
    \PackageError{the-very-soil}{Missing plate #1}%
      {Each enabled Part plate unit must contain exactly four page PDFs.}%
  }%
}

\newcommand{\includepartplates}[1]{%
  \foreach \n in {01,02,03,04}{%
    \BookRequireSinglePagePlate{assets/plates/#1/page-\n.pdf}%
  }%
  \foreach \n in {01,02,03,04}{%
    \includepdf[
      pages=1,
      noautoscale=true,
      trim=3mm 3mm 3mm 3mm,
      clip,
      pagecommand={\thispagestyle{empty}}
    ]{assets/plates/#1/page-\n.pdf}%
  }%
}
